    <h3 class="pub-section-title">Molecular Biosystems — Practical Class</h3>
    <div class="pub-meta">Molecular Biosystems course (institution not stated on the archived page) · Undated (archived)</div>
    <p class="software-note">
      Exercises: (1) Needleman–Wunsch dynamic-programming alignment by hand,
      including the semi-global variant; (2) ClustalOmega and LocARNA
      structural alignments of let-7 and tRNA sequences; (3) further work on
      an upstream-regions dataset.
    </p>
    <aside class="context-note" style="margin-top: 1.2rem;">
      <div class="context-note-label">Provenance note</div>
      <p>
        The archived page names no institution and no date; its dataset links
        point at my INESC-ID / KDBIO hosting (kdbio.inesc-id.pt). The course
        is filed here, in the teaching archive, without firmer provenance —
        the archive itself does not settle where or when it was taught. The
        sequence datasets the exercises worked with (pre-let-7, tRNAs,
        upstream regions, binding sites, hairpins) were part of the
        practicals but are not reproduced on this page.
      </p>
    </aside>

(function () {
  var viewers = document.querySelectorAll('[data-pdf-viewer]');
  Array.prototype.forEach.call(viewers, function (viewer) {
    var frame = viewer.querySelector('[data-pdf-frame]');
    var openLink = viewer.querySelector('[data-pdf-open]');
    var docs = viewer.querySelectorAll('.pdf-doc');
    if (!frame) { return; }
    Array.prototype.forEach.call(docs, function (doc) {
      doc.addEventListener('click', function () {
        Array.prototype.forEach.call(docs, function (other) {
          other.classList.remove('is-active');
          other.setAttribute('aria-pressed', 'false');
        });
        doc.classList.add('is-active');
        doc.setAttribute('aria-pressed', 'true');
        var src = doc.getAttribute('data-pdf-src');
        var title = doc.getAttribute('data-pdf-title');
        frame.setAttribute('src', src);
        frame.setAttribute('title', title);
        if (openLink) {
          openLink.setAttribute('href', src);
          openLink.textContent = 'Open ' + title + ' directly ↗';
        }
      });
    });
  });
})();
</script>
